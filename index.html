async function addCourt(isOwner){

  const p=isOwner?'oa':'aa';

  const name=document.getElementById(p+'N')?.value.trim();const sport=document.getElementById(p+'S')?.value;const surface=document.getElementById(p+'Surf')?.value.trim();const price=parseFloat(document.getElementById(p+'P')?.value);

  if(!name||!price){toast('Fill in all fields','err');return}

  const{data:court,error}=await sb.from('courts').insert({name,sport,surface:surface||'Standard',price_per_hour:price,icon:ICONS[sport]||'🏟️',owner_id:isOwner?CU.id:null,featured:false,photo_urls:[]}).select().single();

  if(error){toast('Failed to add court','err');return}

  if(isOwner&&pendingCourtPhotos.length){const urls=await uploadCourtPhotos(pendingCourtPhotos,court.id);if(urls.length)await sb.from('courts').update({photo_urls:urls}).eq('id',court.id);pendingCourtPhotos=[];}

  [p+'N',p+'Surf',p+'P'].forEach(id=>{const el=document.getElementById(id);if(el)el.value=''});

  closeModal(isOwner?'oAddModal':'aAddModal');toast('Court added!','ok');

async function saveCourt(){

  const id=parseInt(document.getElementById('ecId')?.value);

  const updates={name:document.getElementById('ecN')?.value.trim(),sport:document.getElementById('ecS')?.value,surface:document.getElementById('ecSurf')?.value.trim(),price_per_hour:parseFloat(document.getElementById('ecP')?.value),icon:ICONS[document.getElementById('ecS')?.value]||'🏟️'};

  const newFiles=pendingEditPhotos.filter(p=>p instanceof File);

  const existPaths=pendingEditPhotos.filter(p=>typeof p==='string');

  if(newFiles.length){const uploaded=await uploadCourtPhotos(newFiles,id);updates.photo_urls=[...existPaths,...uploaded].slice(0,4);}

  else if(existPaths.length)updates.photo_urls=existPaths;

  const{error}=await sb.from('courts').update(updates).eq('id',id);

  if(!error){closeModal('editModal');toast('Court updated!','ok');pendingEditPhotos=[];if(CU?.role==='owner')renderOwnerCourts();else renderCtTbl();}

  else toast('Failed to update','err');

}
